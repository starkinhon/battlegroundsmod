# Lote 4 (Byakuya Z vivo, Shikamaru sombras vivas, Ino aura própria): prints e quadros pra comparar com as prévias.
# gradlew runClient -Pmundo=SoulsDev -Proteiro=lote4-subalmas

/devc fechar
/dev preparar
/gamerule soulsBlockDestruction false
/gamemode creative @s
/devc hud off
/devc fov 70
/soul semcooldown
espera 20

# ---------- Byakuya: Z vivo (alvo andando solto) ----------
/soul escolher byakuya
espera 10
/dev invocar minecraft:zombie 12 solto
/dev passivo @e[type=minecraft:zombie]
/dev imortal @e[type=minecraft:zombie]
espera 20
/devc vista terceira
/devc camera lado 9 3
/devc mirar seguir
/devc gravar lote4/byakuya_z 2
/devc tecla z 100
espera 30
/devc gravar parar
/devc print lote4/byakuya_z_fim
/devc mirar parar
/kill @e[type=minecraft:zombie]
espera 10

# ---------- Byakuya: Z sem alvo (rio reto) ----------
/devc olhar 0 0
/devc gravar lote4/byakuya_z_reto 3
/devc tecla z 40
espera 20
/devc gravar parar
espera 20

# ---------- Shikamaru: Z Kagemane, Z de novo Kubishibari ----------
/soul escolher shikamaru
espera 10
/dev invocar minecraft:zombie 10
/dev imortal @e[type=minecraft:zombie]
espera 10
/devc camera duelo 10 4
/dev mirar @e[type=minecraft:zombie,limit=1,sort=nearest]
/devc gravar lote4/shika_z 2
/devc tecla z
espera 50
/devc print lote4/shika_z_preso
espera 10
/devc tecla z
espera 40
/devc print lote4/shika_kubishibari
/devc gravar parar
/devc tecla z
espera 30
/kill @e[type=minecraft:zombie]
espera 10

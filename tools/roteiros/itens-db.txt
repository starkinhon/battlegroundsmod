# Itens DB (PR #299): Senzu, Scouter e Radar com modelo 3D e ícone 32x32.
# Prints em _claude_shots/itensdb: inventário, hotbar, mão (1ª e 3ª pessoa), Scouter na cabeça, chão e quadro.
# Rodar: gradlew runClient -Pmundo=SoulsDev -Proteiro=itens-db

/devc fechar
/dev preparar
/devc janela 1600 900
/devc fov 70
/kill @e[type=!player]
/clear @s
/item replace entity @s hotbar.0 with battlegroundsmod:senzu
/item replace entity @s hotbar.1 with battlegroundsmod:scouter
/item replace entity @s hotbar.2 with battlegroundsmod:radar_dragao
/item replace entity @s hotbar.3 with battlegroundsmod:esfera_dragao_4
/item replace entity @s inventory.0 with battlegroundsmod:senzu 16
/item replace entity @s inventory.1 with minecraft:diamond_sword
/item replace entity @s inventory.2 with minecraft:compass
/gamemode survival
/tp @s ~ ~ ~ 180 10
espera 40

# Inventário aberto (ícones 32x32 ao lado de itens vanilla)
/devc tecla e 2
espera 20
/devc print itensdb/01_inventario
/devc fechar
espera 10

/summon item ~ ~ ~-2 {Item:{id:"battlegroundsmod:scouter",count:1},PickupDelay:32767,Age:-32768}

/summon item_frame ~ ~1 ~-3 {Facing:3b,Fixed:1b,Item:{id:"battlegroundsmod:scouter",count:1}}

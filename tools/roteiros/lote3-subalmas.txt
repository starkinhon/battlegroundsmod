# Lote 3 das subalmas (Nobara, Miwa, Inumaki): grava cada skill pra comparar com a prévia.
# Rodar: gradlew runClient -Pmundo=SoulsDev -Proteiro=lote3-subalmas
# Quadros em _claude_shots/lote3/<skill>/ (um PNG a cada 2 quadros de tela).

/devc fechar
/dev soltar
/kill @e[type=!player]
/dev preparar
/gamerule soulsBlockDestruction false
/devc hud off
/devc fov 70
/soul semcooldown
espera 20

# ================= NOBARA =================
/soul escolher nobara

/dev invocar minecraft:zombie 9
/dev imortal @e[type=minecraft:zombie,limit=1,sort=nearest]

/devc camera duelo 9 1.6
espera 10

# Z: 3 pregos cravam no zumbi (impacto ao cravar)

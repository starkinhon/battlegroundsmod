# Lote 3 das subalmas (Nobara, Miwa, Inumaki): grava cada skill pra comparar com a prévia.
# Rodar: gradlew runClient -Pmundo=SoulsDev -Proteiro=lote3-subalmas
# Quadros em _claude_shots/lote3/<skill>/ (um PNG a cada 2 quadros de tela).

/devc fechar
/dev soltar
/kill @e[type=!player]
/dev preparar
/gamerule soulsBlockDestruction false
/gamemode survival @s
/effect give @s minecraft:resistance 99999 4 true
/effect give @s minecraft:saturation 99999 1 true
/devc hud off
/devc fov 70
/soul semcooldown
espera 20

# ================= NOBARA =================
/soul escolher nobara
/item replace entity @s hotbar.0 with battlegroundsmod:nobara_hammer
/item replace entity @s weapon.mainhand with battlegroundsmod:nobara_hammer

/dev invocar minecraft:zombie 9
/item replace entity @e[type=minecraft:zombie,limit=1,sort=nearest] armor.head with minecraft:chainmail_helmet
/dev imortal @e[type=minecraft:zombie,limit=1,sort=nearest]
espera 10
/devc camera duelo 9 1.6
espera 10

# Criaturas Dragon Ball (PR #317). Rodar: gradlew runClient -Pmundo=SoulsDev -Proteiro=db-mobs-luta
/devc fechar
/dev preparar
/devc hud off
/devc fov 70
/dev soltar
/kill @e[type=!player]
/tp @s 0.5 -60 0.5 180 0
espera 20

/gamemode survival
/dev imortal @s
/effect give @s minecraft:resistance 100000 2 true
/effect give @s minecraft:regeneration 100000 3 true
/effect give @s minecraft:saturation 100000 1 true
espera 10

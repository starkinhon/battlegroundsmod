# Só o kit do Nanami (pedaço do subalmas-lote2), destruição desligada.
# Rodar:  gradlew runClient -Pmundo=SoulsDev -Proteiro=nanami-kit
/devc fechar
/dev preparar
/devc hud off
/devc fov 70
/gamerule soulsBlockDestruction false
/kill @e[type=!player]
/fill -20 -61 20 20 -61 100 minecraft:grass_block
/fill -20 -60 20 20 -50 79 minecraft:air
/fill -20 -60 80 20 -50 100 minecraft:air
/soul semcooldown true
# ================= NANAMI =================
/dev soltar
/kill @e[type=!player]
/tp @s 0 -60 30 180 0
/soul escolher nanami
# Z e C precisam do facão na mão principal (WeaponSkills)
/item replace entity @s weapon.mainhand with battlegroundsmod:nanami_blade
/dev invocar minecraft:husk 6
/dev imortal @e[type=minecraft:husk,limit=1,sort=nearest]
espera 20
/devc camera duelo 10 2.5
/dev mirar @e[type=minecraft:husk,limit=1,sort=nearest]
/devc marca NANAMI Z
/devc gravar lote2/nanami-z 6
/devc tecla z
espera 20
/devc print lote2/nanami-z-regua
/devc tecla z
espera 30
/devc gravar parar
/devc print lote2/nanami-z-fim

/devc marca NANAMI C
/dev mirar @e[type=minecraft:husk,limit=1,sort=nearest]
/devc gravar lote2/nanami-c 6
/devc tecla c

/devc gravar parar
/devc print lote2/nanami-c-fim

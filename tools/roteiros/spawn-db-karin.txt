difficulty normal
gamerule doMobSpawning true
gamerule doDaylightCycle false

execute at @s run locate biome battlegroundsmod:terra_sagrada_de_karin

execute if entity @e[type=battlegroundsmod:raptor,distance=..80]
execute if entity @e[type=battlegroundsmod:tiranossauro,distance=..80]
execute if entity @e[type=battlegroundsmod:soldado_red_ribbon,distance=..80]
execute if entity @e[type=battlegroundsmod:bora,distance=..120]

espera 1200

execute if entity @e[type=battlegroundsmod:tiranossauro,distance=..80]
